cmake_minimum_required(VERSION 3.25)
project(DynamicBottleLiquidsSE VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
include(FetchContent)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
option(DBL_SKYRIM_VR "Build the Skyrim VR 1.4.15 DLL; OFF builds the upstream SE/AE DLL" ON)
if(DBL_SKYRIM_VR)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR ON CACHE BOOL "" FORCE)
else()
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
endif()
find_package(nlohmann_json CONFIG REQUIRED)
find_package(minhook CONFIG REQUIRED)
FetchContent_Declare(dbl_commonlib URL https://codeload.github.com/Zzyxz/CommonLibSSE-NG/zip/d4ee41104c391ac4e19093c97c3a76138bf6b55a SOURCE_SUBDIR extern/CommonLibSSE)
FetchContent_MakeAvailable(dbl_commonlib)
add_library(SkyrimBottlePrototype SHARED Plugin.cpp PerformanceUI.cpp core/Simulation.cpp)
target_include_directories(SkyrimBottlePrototype PRIVATE core)
target_compile_definitions(SkyrimBottlePrototype PRIVATE DBL_FRAMEWORK_UI NOMINMAX WIN32_LEAN_AND_MEAN DBL_FIXED_SETTINGS)
if(DBL_SKYRIM_VR)
target_compile_definitions(SkyrimBottlePrototype PRIVATE DBL_SKYRIM_VR)
option(DBL_VR_DIAGNOSTICS "Log Skyrim VR draw-interception diagnostics" OFF)
if(DBL_VR_DIAGNOSTICS)
target_compile_definitions(SkyrimBottlePrototype PRIVATE DBL_VR_DIAGNOSTICS)
endif()
option(DBL_VR_TIMING "Test builds: log DBL timing every 10 s without the overlay" OFF)
if(DBL_VR_TIMING)
target_compile_definitions(SkyrimBottlePrototype PRIVATE DBL_VR_TIMING)
endif()
else()
target_compile_definitions(SkyrimBottlePrototype PRIVATE DBL_STEAM_17104 DBL_STEAM_UNIFIED)
endif()
target_precompile_headers(SkyrimBottlePrototype PRIVATE <SKSE/Impl/PCH.h>)
target_link_libraries(SkyrimBottlePrototype PRIVATE CommonLibSSE::CommonLibSSE nlohmann_json::nlohmann_json minhook::minhook d3d11 d3dcompiler)
