cmake_minimum_required(VERSION 3.25)
project(DynamicBottleLiquidsSE VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
include(FetchContent)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(minhook CONFIG REQUIRED)
FetchContent_Declare(dbl_commonlib URL https://codeload.github.com/Zzyxz/CommonLibSSE-NG/zip/d4ee41104c391ac4e19093c97c3a76138bf6b55a SOURCE_SUBDIR extern/CommonLibSSE)
FetchContent_MakeAvailable(dbl_commonlib)
add_library(SkyrimBottlePrototype SHARED Plugin.cpp PerformanceUI.cpp core/Simulation.cpp)
target_include_directories(SkyrimBottlePrototype PRIVATE core)
target_compile_definitions(SkyrimBottlePrototype PRIVATE DBL_FRAMEWORK_UI NOMINMAX WIN32_LEAN_AND_MEAN DBL_FIXED_SETTINGS DBL_STEAM_17104 DBL_STEAM_UNIFIED)
target_precompile_headers(SkyrimBottlePrototype PRIVATE <SKSE/Impl/PCH.h>)
target_link_libraries(SkyrimBottlePrototype PRIVATE CommonLibSSE::CommonLibSSE nlohmann_json::nlohmann_json minhook::minhook d3d11 d3dcompiler)
